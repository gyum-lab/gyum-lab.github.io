% ---------------- Research interests ----------------
\section{Research Interests}
\begin{itemize}
  \item \textbf{Physical design methodologies for 3D integration} --- hybrid-bonded F2F 3D ICs
        (tier partitioning, 3D via legalization), unified flows for homogeneous and heterogeneous 3D ICs,
        monolithic 3D ICs (thermal-aware placement, cross-tier routability)
  \item \textbf{Backside design methodologies for advanced technology nodes} --- backside power delivery
        network (BS-PDN) optimization for F2F 3D ICs, backside clock distribution network (BS-CDN) synthesis
  \item \textbf{Machine learning for EDA (ML-EDA)} --- reinforcement learning for routing congestion mitigation
        and 3D via legalization, graph neural networks for physical design, learning-based prediction for
        placement refinement and thermal analysis
\end{itemize}
